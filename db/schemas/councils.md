create table public.councils (
  council_id integer generated by default as identity not null,
  name text not null,
  resolution_count integer not null default 0,
  is_main boolean null default false,
  constraint councils_pkey primary key (council_id),
  constraint councils_name_key unique (name)
) TABLESPACE pg_default;

create unique INDEX IF not exists unique_main_council on public.councils using btree (is_main) TABLESPACE pg_default
where
  (is_main = true);
